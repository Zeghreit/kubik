7113: function buildSymmetryMap(obj, axis) {
7114:   const topo = obj.mesh.userData.topo;
7115:   if (!topo || !topo.logicalCount) return new Map();
7116: 
7117:   /* Nearest-within-tolerance rather than an exact rounded-key lookup. The
7118:      key version needed two vertices to agree to 0.0001 in every coordinate,
7119:      so the tiniest drift from an earlier edit silently produced no partners
7120:      at all - and symmetry then did nothing, with no way to tell it apart
7121:      from working. Tolerance scales with the object so it holds for large
7122:      and small models alike. */
7123:   const pts = [];
7124:   for (let l = 0; l < topo.logicalCount; l++) pts.push(logicalPos(obj, l));
7125:   const size = new THREE.Box3().setFromPoints(pts).getSize(new THREE.Vector3());
7126:   const tol = Math.max(size.x, size.y, size.z, 1) * 1e-3;
7127:   const tol2 = tol * tol;
7128: 
7129:   // Reflect about the captured plane, not local zero: x' = 2*offset - x.
7130:   const off = symmetryPlane(obj, axis).offset;
7131:   const map = new Map();
7132:   const m = new THREE.Vector3();
7133:   for (let l = 0; l < topo.logicalCount; l++) {
7134:     m.copy(pts[l]);
7135:     m[axis] = 2 * off - m[axis];
7136:     let best = -1, bestD = tol2;
7137:     for (let k = 0; k < pts.length; k++) {
7138:       const d = m.distanceToSquared(pts[k]);
7139:       if (d < bestD) { bestD = d; best = k; }
7140:     }
7141:     if (best >= 0) map.set(l, best);
7142:   }
